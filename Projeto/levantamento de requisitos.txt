

6. Instruções SQL

CREATE TABLE "clientes" (
	"cpf"	TEXT NOT NULL,
	"nome_cliente"	TEXT NOT NULL
);

CREATE TABLE "produtos" (
	"cod_barras"	INTEGER NOT NULL,
	"nome_produto"	TEXT NOT NULL,
	"estoque_produto"	INTEGER NOT NULL DEFAULT 0,
	PRIMARY KEY("cod_barras")
);

CREATE TABLE "vendas" (
	"id_vendas"	INTEGER NOT NULL,
	"nr_venda"	INTEGER NOT NULL,
	"status_vendas"	TEXT NOT NULL,
	"qtd_produto"	INTEGER NOT NULL DEFAULT 0,
	"id_cliente"	INTEGER NOT NULL,
	"cod_barras"	INTEGER NOT NULL,
	PRIMARY KEY("id_vendas" AUTOINCREMENT)
);
